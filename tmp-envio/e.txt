// (excluir_minha_conta) e (3) manda o e-mail de despedida, pelo Gmail cafeedclub@gmail.com, com resposta para contato@cafeedclub.com.
//
// Segredos (Supabase → Edge Functions → Secrets):
//   GMAIL_USUARIO   = cafeedclub@gmail.com (o Zoho grátis não tem SMTP; quem responde cai em contato@cafeedclub.com)
//   GMAIL_SENHA_APP = a senha de app do Google (16 letras)
// Não usa a chave service_role: age com o login da própria pessoa.
//
// O e-mail (MOLDE) é gerado por scripts/gerar-emails.py a partir de email/conta-excluida.html.

const CORS = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
  'Access-Control-Allow-Methods': 'POST, OPTIONS',
};

function resposta(corpo: unknown, status = 200) {
  return new Response(JSON.stringify(corpo), { status, headers: { ...CORS, 'Content-Type': 'application/json' } });
}

function escapar(texto: string) {
  return texto.replace(/[&<>"']/g, (c) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' })[c]!);
}

Deno.serve(async (req) => {
  if (req.method === 'OPTIONS') return new Response('ok', { headers: CORS });
  if (req.method !== 'POST') return resposta({ erro: 'Use POST' }, 405);

  const autorizacao = req.headers.get('Authorization') ?? '';
  if (!autorizacao.startsWith('Bearer ')) return resposta({ erro: 'Não autenticado' }, 401);
  // a chave pública vem do próprio app (cabeçalho apikey); se faltar, usa a do projeto
  const chave = req.headers.get('apikey') ?? Deno.env.get('SUPABASE_ANON_KEY') ?? '';
  const supabase = createClient(Deno.env.get('SUPABASE_URL')!, chave, {
    global: { headers: { Authorization: autorizacao } },
  });

  // 1) quem está pedindo
  const { data: dados, error: erroUsuario } = await supabase.auth.getUser();
  const usuario = dados?.user;
  if (erroUsuario || !usuario) return resposta({ erro: 'Não autenticado' }, 401);
  const email = usuario.email ?? '';
  let nome = String(usuario.user_metadata?.nome ?? '').trim();
  if (!nome) {
    const { data: perfil } = await supabase.from('perfis').select('nome').eq('id', usuario.id).maybeSingle();
    nome = String(perfil?.nome ?? '').trim();
  }
  nome = nome.split(/\s+/)[0] ?? '';

  // 2) apaga a conta (mesma função que o app usava)
  const { error: erroExcluir } = await supabase.rpc('excluir_minha_conta');
  if (erroExcluir) return resposta({ erro: erroExcluir.message }, 400);

  // 3) e-mail de despedida (se falhar, a conta já foi excluída do mesmo jeito)
  let emailEnviado = false;
  const remetente = Deno.env.get('GMAIL_USUARIO');
  const senha = Deno.env.get('GMAIL_SENHA_APP');
  if (email && remetente && senha && MOLDE) {
    try {
      // idioma gravado no cadastro ('pt' | 'en' | 'es'); contas antigas não têm, e nelas vale o campo ingles
      const idioma = usuario.user_metadata?.idioma ?? (usuario.user_metadata?.ingles ? 'en' : 'pt');
      const html = (idioma === 'es' ? MOLDE_ES : idioma === 'en' ? MOLDE_EN : MOLDE).replaceAll('{{nome}}', nome ? ', ' + escapar(nome) : '').replaceAll('{{email}}', escapar(email));
      const correio = nodemailer.createTransport({ host: 'smtp.gmail.com', port: 465, secure: true, auth: { user: remetente, pass: senha } });
      await correio.sendMail({
        from: `Cafeed <${remetente}>`,
        replyTo: 'Cafeed <contato@cafeedclub.com>',
        to: email,
        subject: idioma === 'es' ? 'Tu cuenta de Cafeed fue eliminada' : idioma === 'en' ? 'Your Cafeed account was deleted' : 'Sua conta do Cafeed foi excluída',
        html,
        text: idioma === 'es'
          ? `¡Hasta la próxima${nome ? ', ' + nome : ''}! Tu cuenta en Cafeed fue eliminada, como pediste. Si quieres volver, solo crea una cuenta nueva en www.cafeedclub.com/app. Si no pediste esto, escribe a contato@cafeedclub.com.`
          : idioma === 'en'
          ? `See you around${nome ? ', ' + nome : ''}! Your Cafeed account was deleted, as you asked. If you want to come back, just create a new account at www.cafeedclub.com. If you did not ask for this, write to contato@cafeedclub.com.`
          : `Até a próxima${nome ? ', ' + nome : ''}! Sua conta no Cafeed foi excluída, como você pediu. Se quiser voltar, é só criar uma conta nova em www.cafeedclub.com/app. Se você não pediu isso, escreva para contato@cafeedclub.com.`,
      });
      emailEnviado = true;
    } catch (e) {
      console.error('e-mail de despedida não saiu:', e);
    }
  }
  return resposta({ ok: true, emailEnviado });
});
