// CAFEED — função "traduzir": o botão "Ver tradução" dos posts, comentários e conversas.
// Recebe { texto, para: 'pt' | 'en' } de quem está logado e devolve { traducao, mesma }.
// Usa a OpenAI (modelo gpt-4o-mini, centavos por mês) com o segredo OPENAI_API_KEY, que precisa da permissão
// "Model capabilities → Chat completions" (além de Moderations). Guarda cada tradução em public.traducoes
// (supabase/traducoes.sql) para não traduzir o mesmo texto de novo. Até 300 traduções por pessoa por dia.
// Ao criar a função no Supabase, desligue "Verify JWT" (a função confere o login sozinha).
import { createClient } from 'npm:@supabase/supabase-js@2.45.4';

const CORS = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
  'Access-Control-Allow-Methods': 'POST, OPTIONS',
};
function resposta(corpo: unknown, status = 200) {
  return new Response(JSON.stringify(corpo), { status, headers: { ...CORS, 'Content-Type': 'application/json' } });
}
async function sha256(s: string) {
  const b = new Uint8Array(await crypto.subtle.digest('SHA-256', new TextEncoder().encode(s)));
  return Array.from(b).map((x) => x.toString(16).padStart(2, '0')).join('');
}
const NOMES = { pt: 'Brazilian Portuguese', en: 'English' } as const;

Deno.serve(async (req) => {
  if (req.method === 'OPTIONS') return new Response('ok', { headers: CORS });
  if (req.method !== 'POST') return resposta({ erro: 'Use POST' }, 405);

  const autorizacao = req.headers.get('Authorization') ?? '';
  if (!autorizacao.startsWith('Bearer ')) return resposta({ erro: 'Não autenticado' }, 401);
  const chave = req.headers.get('apikey') ?? Deno.env.get('SUPABASE_ANON_KEY') ?? '';
  const comoPessoa = createClient(Deno.env.get('SUPABASE_URL')!, chave, { global: { headers: { Authorization: autorizacao } } });
  const { data: dados, error: erroUsuario } = await comoPessoa.auth.getUser();
  const usuario = dados?.user;
  if (erroUsuario || !usuario) return resposta({ erro: 'Não autenticado' }, 401);

  let texto = '';
  let para: 'pt' | 'en' = 'en';
  try {
    const corpo = await req.json();
    texto = typeof corpo?.texto === 'string' ? corpo.texto.trim() : '';
    para = corpo?.para === 'pt' ? 'pt' : 'en';
  } catch {
    return resposta({ erro: 'Corpo inválido' }, 400);
  }
  if (!texto || texto.length > 1500) return resposta({ erro: 'Texto vazio ou longo demais' }, 400);

  const banco = createClient(Deno.env.get('SUPABASE_URL')!, Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!, { auth: { persistSession: false } });
  const id = await sha256(para + '|' + texto);
  const { data: guardada } = await banco.from('traducoes').select('traducao, mesma').eq('chave', id).maybeSingle();
  if (guardada) return resposta({ traducao: guardada.traducao, mesma: guardada.mesma });

  const { data: pode } = await banco.rpc('contar_traducao', { p_user: usuario.id });
  if (pode === false) return resposta({ erro: 'Muitas traduções hoje. Tente amanhã.' }, 429);

  const chaveOpenAi = Deno.env.get('OPENAI_API_KEY');
  if (!chaveOpenAi) return resposta({ erro: 'Tradução indisponível' }, 503);
  const r = await fetch('https://api.openai.com/v1/chat/completions', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', Authorization: 'Bearer ' + chaveOpenAi },
    body: JSON.stringify({
      model: 'gpt-4o-mini',
      temperature: 0.2,
      messages: [
        {
          role: 'system',
          content:
            `You translate short posts from a coffee-shop social app into ${NOMES[para]}. ` +
            'Keep the tone, emojis, line breaks, @mentions, #hashtags and names of places exactly as they are. ' +
            'Reply with the translation only, no quotes or notes. ' +
            `If the text is already in ${NOMES[para]}, reply with exactly: ===SAME===`,
        },
        { role: 'user', content: texto },
      ],
    }),
  });
  if (!r.ok) {
    console.error('[traduzir] OpenAI respondeu', r.status, (await r.text()).slice(0, 300));
    return resposta({ erro: 'Tradução indisponível agora' }, 503);
  }
  const saida = String((await r.json())?.choices?.[0]?.message?.content ?? '').trim();
  if (!saida) return resposta({ erro: 'Tradução indisponível agora' }, 503);
  const mesma = saida === '===SAME===';
  const traducao = mesma ? texto : saida.slice(0, 3000);
  await banco.from('traducoes').upsert({ chave: id, idioma: para, traducao, mesma }, { onConflict: 'chave' });
  return resposta({ traducao, mesma });
});
