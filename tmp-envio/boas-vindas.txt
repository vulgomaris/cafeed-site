const CORS = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
  'Access-Control-Allow-Methods': 'POST, OPTIONS',
};

function resposta(corpo: unknown, status = 200) {
  return new Response(JSON.stringify(corpo), { status, headers: { ...CORS, 'Content-Type': 'application/json' } });
}

function escapar(texto: string) {
  return texto.replace(/[&<>"']/g, (c) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' })[c]!);
}

Deno.serve(async (req) => {
  if (req.method === 'OPTIONS') return new Response('ok', { headers: CORS });
  if (req.method !== 'POST') return resposta({ erro: 'Use POST' }, 405);

  const autorizacao = req.headers.get('Authorization') ?? '';
  if (!autorizacao.startsWith('Bearer ')) return resposta({ erro: 'Não autenticado' }, 401);
  const chave = req.headers.get('apikey') ?? Deno.env.get('SUPABASE_ANON_KEY') ?? '';
  const supabase = createClient(Deno.env.get('SUPABASE_URL')!, chave, {
    global: { headers: { Authorization: autorizacao } },
  });

  // 1) quem está pedindo
  const { data: dados, error: erroUsuario } = await supabase.auth.getUser();
  const usuario = dados?.user;
  if (erroUsuario || !usuario) return resposta({ erro: 'Não autenticado' }, 401);

  // Teste: { "teste": true } manda o e-mail de novo para quem pediu, só se for admin (não marca nada na conta)
  let teste = false;
  try {
    teste = (await req.json())?.teste === true;
  } catch {
    /* corpo vazio */
  }
  if (teste) {
    const { data: admin } = await supabase.rpc('sou_admin');
    if (admin !== true) return resposta({ erro: 'Só para admins.' }, 403);
  }

  // 2) cada conta recebe as boas-vindas uma vez só (e só logo depois de criada)
  if (!teste && usuario.user_metadata?.boas_vindas_em) return resposta({ ok: true, emailEnviado: false, motivo: 'já enviado' });
  const criadaHa = Date.now() - new Date(usuario.created_at).getTime();
  if (!teste && criadaHa > 24 * 60 * 60 * 1000) return resposta({ ok: true, emailEnviado: false, motivo: 'conta antiga' });

  const email = usuario.email ?? '';
  const nome = String(usuario.user_metadata?.nome ?? '').trim().split(/\s+/)[0] ?? '';
  const remetente = Deno.env.get('GMAIL_USUARIO');
  const senha = Deno.env.get('GMAIL_SENHA_APP');
  if (!email || !remetente || !senha || !MOLDE) return resposta({ ok: false, emailEnviado: false }, 500);

  // 3) marca antes de mandar, para dois toques seguidos não mandarem dois e-mails
  if (!teste) {
    const { error: erroMarca } = await supabase.auth.updateUser({ data: { boas_vindas_em: new Date().toISOString() } });
    if (erroMarca) return resposta({ erro: erroMarca.message }, 400);
  }
  try {
    const en = Boolean(usuario.user_metadata?.ingles); // quem mora fora do Brasil recebe em inglês
    const html = (en ? MOLDE_EN : MOLDE).replaceAll('{{nome}}', nome ? ', ' + escapar(nome) : '').replaceAll('{{email}}', escapar(email));
    const correio = nodemailer.createTransport({ host: 'smtp.gmail.com', port: 465, secure: true, auth: { user: remetente, pass: senha } });
    await correio.sendMail({
      from: `Cafeed <${remetente}>`,
      replyTo: 'Cafeed <contato@cafeedclub.com>',
      to: email,
      subject: en ? 'Welcome to Cafeed!' : 'Boas-vindas ao Cafeed!',
      html,
      text: en
        ? `Welcome to Cafeed${nome ? ', ' + nome : ''}! We are so glad you pulled up a chair. Your account is ready: www.cafeedclub.com`
        : `Boas-vindas ao Cafeed${nome ? ', ' + nome : ''}! Que bom que você puxou uma cadeira para sentar com a gente. Sua conta já está pronta: www.cafeedclub.com/app`,
    });
    return resposta({ ok: true, emailEnviado: true });
  } catch (e) {
    console.error('e-mail de boas-vindas não saiu:', e);
    return resposta({ ok: false, emailEnviado: false }, 500);
  }
});
